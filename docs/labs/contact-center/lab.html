    credentials: {
        access_token: 'your_token'

Step 2: Agent Registration

// Simple Usage: Register agent
const response = await window.webex.cc.register();

Step 3: Station Login

// Simple Usage: Login to station
await window.webex.cc.stationLogin({
    teamId: 'team_id',
    loginOption: 'BROWSER',
    dialNumber: ''  // Required for AGENT_DN/EXTENSION
});

Step 4: Agent State Management

// Simple Usage: Set agent state
await window.webex.cc.setAgentState({
    state: 'WellbeingBreak',
    auxCodeId: 'aux_code_id'
});

Step 5: Task Management

// Simple Usage: Handle tasks
window.webex.cc.on('task:new', (task) => {
    await task.accept();
});

Step 6: Call Controls

// Simple Usage: Voice controls
await task.hold();    // Hold call
await task.resume();  // Resume call
task.toggleMute();    // Toggle mute
