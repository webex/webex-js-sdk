Step 2: Agent Registration

// Initialize and Register with Contact Center SDK
async function initializeContactCenter() {
    // 1. Initialize SDK with auth token
    const webex = window.webex = Webex.init({
        credentials: {
            access_token: 'your_token'
        }
    });

    try {
        // 2. Register agent and get profile
        const agentProfile = await webex.cc.register();
        console.log('Agent registered:', agentProfile);

        // 3. Store important agent data
        window.wrapupCodes = agentProfile.wrapupCodes;
        window.teams = agentProfile.teams;
        window.loginVoiceOptions = agentProfile.loginVoiceOptions;

        // 4. Initialize task event handlers
        initializeTaskEvents();
        
        // 5. Handle agent state changes
        webex.cc.on('agent:stateChange', ({state, auxCodeId}) => {
            console.log('Agent state changed:', state);
            updateAgentStateUI(state, auxCodeId);
        });
        
        return agentProfile;
    } catch (error) {
        console.error('Registration failed:', error);
        throw error;
    }
}

// Deregister agent
async function deregisterAgent() {
    try {
        await webex.cc.deregister();
        console.log('Agent deregistered successfully');
    } catch (error) {
        console.error('Deregistration failed:', error);
        throw error;
    }
}

Note: Contact Center SDK automatically handles reconnection and session management.

// Station Login using Contact Center SDK
async function initializeStationLogin(agentProfile) {
    // 1. Get available teams from registration
    const teams = agentProfile.teams;
    console.log('Available teams:', teams);

    // 2. Get login options
    const loginOptions = agentProfile.loginVoiceOptions;
    console.log('Login options:', loginOptions);

    // 3. Perform station login
    try {
        const response = await webex.cc.stationLogin({
            teamId: selectedTeam.id,
            loginOption: selectedOption, // 'BROWSER', 'AGENT_DN', 'EXTENSION'
            dialNumber: selectedOption === 'BROWSER' ? '' : dialNumber
        });
        console.log('Station login successful:', response);

        // 4. Listen for station events
        webex.cc.on('agent:stationLoginSuccess', handleStationLogin);
        webex.cc.on('agent:reloginSuccess', handleStationRelogin);
    } catch (error) {
        console.error('Station login failed:', error);
        handleLoginError(error);
    }
}

// Agent State Management using Contact Center SDK
async function initializeAgentState() {
    // Get agent's current state
    const agentProfile = await webex.cc.getAgentState();
    console.log('Current state:', agentProfile.state);

// Set agent state with reason code
async function setAgentState(state, auxCodeId) {
    try {
        await webex.cc.setAgentState({
            state,              // e.g., 'Available', 'WellbeingBreak'

    }
}

Step 5: Task Management

// Task Event Handling using Contact Center SDK
const initializeTaskEvents = () => {
    // Contact Center Events
    webex.cc.on('task:incoming', (task) => {
        console.log('New task received:', task.data.interactionId);
    });
    
    webex.cc.on('task:hydrate', (task) => {
        console.log('Task hydrated:', task.data.interactionId);
        handleTaskHydrate(task);
    });
    
    webex.cc.on('agent:stateChange', (data) => {
        console.log('Agent state changed:', data.state);
        updateAgentStateUI(data);
    });
    
    webex.cc.on('agent:multiLogin', (data) => {
        if (data.type === 'AgentMultiLoginCloseSession') {
            console.warn('Multiple Agent Login Session Detected!');
        }
    });
    
    webex.cc.on('agent:reloginSuccess', (data) => {
        console.log('Agent re-login successful:', data);
        updateAgentLoginUI(data);
    });
    
    webex.cc.on('agent:stationLoginSuccess', (data) => {
        console.log('Agent station-login success:', data);
        updateStationLoginUI(data);
    });

    // Task Events
    task.on('task:assigned', (task) => {
        console.log('Task assigned:', task.data.interactionId);
        handleTaskAssigned(task);
    });

    task.on('task:media', (track) => {
        handleMediaTrack(track);
    });

    task.on('task:end', (task) => {
        console.log('Task ended:', task.data.interactionId);
        handleTaskEnd(task);
    });

    task.on('task:hold', (task) => {
        console.log('Call put on hold:', task.data.interactionId);
        updateHoldUI(task);
    });

    // Consult Events
    // Note: Consult related events and handlers are managed in task-consult.js
    // Available consult events: 
    // - task:consultCreated - When a consult is created
    // - task:offerConsult - When receiving a consult offer
    // - task:consultAccepted - When consult is accepted
    // - task:consulting - During active consultation
    // - task:consultQueueFailed - When queue consult fails
    // - task:consultQueueCancelled - When queue consult is cancelled
    // - task:consultEnd - When consultation ends
    // See task-consult.js for detailed implementation
    setupConsultEventListeners(task);

    task.on('task:rejected', (reason) => {
        console.warn('Task rejected:', reason);
        handleTaskRejected(reason);
    });
};

// Task List Management
function updateTaskList(tasks) {
    const taskList = document.getElementById('taskList');
    taskList.innerHTML = '';
    tasks.forEach(task => {
        const taskItem = document.createElement('div');
        taskItem.className = 'task-item' + (task === window.currentTask ? ' selected' : '');
        
        const details = document.createElement('div');
        details.className = 'task-details';
        details.innerHTML = `
            
ID: ${task.data.interactionId}

            
Type: ${task.data.interaction.mediaType}

            
State: ${task.data.state}

        `;
        
        taskItem.appendChild(details);
        taskList.appendChild(taskItem);
        
        // Make task selectable
        taskItem.addEventListener('click', () => {
            document.querySelectorAll('.task-item').forEach(item => 
                item.classList.remove('selected'));
            taskItem.classList.add('selected');
            window.currentTask = task;
            enableCallControls(task);
        });
    });
}

// Wrapup handling
function updateWrapupCodes(codes) {
    const select = document.getElementById('wrapup-codes');
    select.innerHTML = '';
    codes.forEach(code => {
        const option = document.createElement('option');
        option.value = code.id;
        option.textContent = code.name;
        select.appendChild(option);
    });
    select.disabled = false;
}

// Hold/Resume Handling
async function handleHoldResume(task) {
    const holdBtn = document.getElementById('btn-hold');
    holdBtn.disabled = true;
    try {
        if (isHold) {
            await task.resume();
            holdBtn.textContent = 'Hold Call';
            isHold = false;
        } else {
            await task.hold();
            holdBtn.textContent = 'Resume Call';
            isHold = true;
        }
    } catch (error) {
        console.error('Hold/Resume failed:', error);
    } finally {
        holdBtn.disabled = false;
    }
}

// Initialize voice task handling
function initializeTaskHandling() {

    webex.cc.on('task:end', handleTaskEnd);
    webex.cc.on('task:hold', handleTaskHold);

// Task End and Wrapup Example
async function handleTaskEnd(taskData) {
    if (taskData.wrapUpRequired) {
        const task = webex.cc.getTask(taskData.interactionId);
        // Enable wrapup with codes from registration
        const wrapUpResponse = await task.wrapup({
            wrapUpReason: 'Resolved',
            auxCodeId: selectedWrapupCode
        });
        console.log('Wrapup completed:', wrapUpResponse);
    }
}

// Hold/Resume Example 
async function handleCallControl(task) {
    try {
        // Hold call
        await task.hold();
        console.log('Call held successfully');

        // Resume call
        await task.resume(); 
        console.log('Call resumed successfully');

        // Toggle mute
        await task.toggleMute();
        console.log('Mute toggled');
    } catch (error) {
        console.error('Call control error:', error);
    }
}

// Consult Transfer Example
async function handleConsultTransfer(task) {
    try {
        // Start consult
        await task.consult({
            destinationType: 'queue',
            to: 'queue_id',
            mediaType: 'telephony'
        });

        // Complete transfer
        await task.consultTransfer();
        console.log('Consult transfer completed');
    } catch (error) {
        console.error('Consult transfer error:', error); 
    }
}
