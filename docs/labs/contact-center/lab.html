Step 2: Agent Registration

After authentication, register the agent to receive tasks and calls.

// Registration Code
async function register(webex) {
    try {
        const response = await webex.cc.register();
        console.log('Registration successful:', response);
        return response;
    } catch (error) {
        console.error('Registration failed:', error);
        throw error;
    }
}

async function deregister(webex) {
    try {
        await webex.cc.deregister();
        console.log('Deregistration successful');
    } catch (error) {
        console.error('Deregistration failed:', error);
        throw error;
    }
}

Step 3: Task Management

Listen for incoming tasks and handle accept/decline operations.

// Task Management Code
function handleIncomingTask(webex, callback) {
    webex.cc.on('task:new', (task) => {
        console.log('New task received:', task);
        callback(task);
    });
}

async function acceptTask(task) {
    try {
        await task.accept();
        console.log('Task accepted successfully');
    } catch (error) {
        console.error('Task accept failed:', error);
        throw error;
    }
}

async function declineTask(task) {
    try {
        await task.decline();
        console.log('Task declined successfully');
    } catch (error) {
        console.error('Task decline failed:', error);
        throw error;
    }
}

Step 4: Digital Channels (Chat / Email)

Initialize and load the IMI Engage widget for handling digital channel interactions.

// Digital Channels Code
function initializeEngageWidget(accessToken) {
    if (window.imiLoaded) return;
    
    const script = document.createElement('script');
    script.src = 'https://wc.imiengage.io/v0.9.11/momentum/js/momentum.js';
    script.onload = () => {
        console.log('IMI Engage script loaded');
        MomentumUI.load({ accessToken });
    };
    document.head.appendChild(script);
    window.imiLoaded = true;
}

Step 5: Call Controls

Manage voice calls with hold/resume and mute/unmute functionality.

// Call Controls Code
let isHold = false;
let isMuted = false;

async function toggleHold(task) {
    try {
        if (!isHold) {
            await task.hold();
            isHold = true;
            console.log('Call held successfully');
        } else {
            await task.resume();
            isHold = false;
            console.log('Call resumed successfully');
        }
    } catch (error) {
        console.error('Hold/Resume operation failed:', error);
        throw error;
    }
}

function toggleMute(task) {
    try {
        task.toggleMute();
        isMuted = !isMuted;
        console.log('Mute toggled successfully, state:', isMuted ? 'muted' : 'unmuted');
    } catch (error) {
        console.error('Mute operation failed:', error);
        throw error;
    }
}

Step 6: Agent State Management

Change the agent's state between Available, Idle, and other states.

// Agent State Code
async function setAgentState(webex, { state, auxCodeId }) {
    try {
        const response = await webex.cc.setAgentState({
            state,
            auxCodeId
        });
        console.log('Agent state set successfully:', state);
        return response;
    } catch (error) {
        console.error('Agent state set failed:', error);
        throw error;
    }
}
